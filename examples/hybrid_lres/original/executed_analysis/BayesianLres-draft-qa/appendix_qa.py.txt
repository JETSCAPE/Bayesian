def read(path):
    return json.loads(path.read_text())


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def arrays(path):
    with np.load(path, allow_pickle=False) as z:
        return {k: z[k].copy() for k in z.files}


def save(path, value):
    path.write_text(json.dumps(value, indent=2, allow_nan=False) + '\n')


def inputs():
    paths = [QUEUE / 'MANIFEST.json', QUEUE / 'covariances.npz', RAW]
    paths += [QUEUE / 'results/all/QA' / name for name in
              ('fivefold_predictions.npz', 'independent_validation.npz',
               'optimizer_and_PC_audit.json', 'validation_summary.json')]
    pins = {str(p): sha(p) for p in paths}
    rows = read(paths[0])['models']['all']['features']
    exp = arrays(paths[1])
    own = [(n, float(a), float(b)) for n, a, b in rows]
    full = [(n, float(a), float(b)) for n, a, b in exp['features']]
    selected = [full.index(row) for row in own]
    raw = arrays(RAW)
    raw_rows = list(zip(raw['observable_names'], raw['xmin'], raw['xmax']))
    ri = [raw_rows.index(row) for row in own]
    cv, hold = arrays(paths[3]), arrays(paths[4])
    assert cv['truth'].shape == (39, 138) and hold['design_ids'].tolist() == [38]
    # Frozen prediction text tables were serialized at finite decimal precision.
    np.testing.assert_allclose(cv['truth'], raw['central_values'][cv['design_ids']][:, ri], rtol=2e-6, atol=1e-12)
    np.testing.assert_allclose(hold['truth'], raw['central_values'][[38]][:, ri], rtol=2e-6, atol=1e-12)
    assert len(set(own)) == len(own) == 138
    indices = [np.array([i for i, r in enumerate(own) if tag in r[0]]) for _, _, tag in GROUPS]
    assert sorted(np.concatenate(indices).tolist()) == list(range(138))
    cexp = exp['covariance_ell_0p2'][np.ix_(selected, selected)]
    return pins, own, raw, ri, cv, hold, indices, cexp

def closure(refine=False):
    from functools import partial
    from types import FunctionType, SimpleNamespace
    import queue_models as api
    import remaining_inference as infer
    from checks import partial_prediction
    ROOT.mkdir(exist_ok=True)
    output_tag = 'closure_grid_181x121' if refine else 'closure_grid'
    assert not (ROOT / (output_tag+'.npz')).exists(), 'Do not repeat a completed grid'
    pins, own, raw, ri, cv, hold, groups, cexp = inputs()
    bank = api.load_bank('all', experimental=False)
    identity = api.model_identity('all')
    assert own == [(n, float(a), float(b)) for n,a,b in bank['features']]
    model = bank['mean_model']
    train = bank['raw']['Design_indices'].astype(int)
    np.testing.assert_array_equal(model.noise, raw['covariance'][train][:, ri][:, :, train][:, :, :, ri])
    cross = raw['covariance'][38, ri][:, train][:, :, ri]
    target = raw['covariance'][38, ri, 38][:, ri]
    exact = partial_prediction(model, hold['design_unit'], np.arange(138), cross[None], target[None])
    np.testing.assert_allclose(exact[0], hold['observed_mean'], atol=2e-10, rtol=1e-9)
    np.testing.assert_allclose(exact[1], hold['observed_covariance'], atol=2e-10, rtol=1e-8)
    # Reuse the exact tested helper bytecode with matrix-chain contraction
    # ordering enabled. The original helper/module and all models stay intact.
    fast_numpy = SimpleNamespace(**(vars(np) | {'einsum': partial(np.einsum, optimize=True)}))
    fast_prediction = FunctionType(partial_prediction.__code__,
        partial_prediction.__globals__ | {'np': fast_numpy})
    check_points = np.vstack((hold['design_unit'], [[.4,.4],[.7,.5]]))
    check_cross = np.broadcast_to(cross, (3,)+cross.shape)
    check_target = np.broadcast_to(target, (3,)+target.shape)
    slow = partial_prediction(model,check_points,np.arange(138),check_cross,check_target)
    fast = fast_prediction(model,check_points,np.arange(138),check_cross,check_target)
    for a,b in zip(fast,slow):
        np.testing.assert_allclose(a,b,atol=2e-10,rtol=1e-8)
    template = arrays(QUEUE / 'results/all/inference/grid.npz')
    # One untouched validation target with experimental covariance, no replica draw.
    stride = 2 if refine else 4
    q, kappa = template['q'][::stride], template['kappa'][::stride]
    def evaluate(q, kappa):
        points = infer.physical_points(q, kappa)
        ll = np.full((len(q),len(kappa)),np.nan)
        if refine:
            previous = arrays(ROOT/'closure_grid.npz')
            old_report = read(ROOT/'CLOSURE.json')
            assert old_report['fit_identity'] == identity and old_report['source_pins'] == pins
            np.testing.assert_array_equal(previous['q'],q[::2])
            np.testing.assert_array_equal(previous['kappa'],kappa[::2])
            ll[::2,::2] = previous['loglike']
        missing = np.flatnonzero(np.isnan(ll.ravel()))
        for start in range(0, len(missing), 64):
            ids = missing[start:start+64]
            batch = points[ids]
            mean, cov = fast_prediction(model, batch, np.arange(138),
                np.broadcast_to(cross, (len(batch),)+cross.shape), np.broadcast_to(target, (len(batch),)+target.shape))
            factor = np.linalg.cholesky(cov + cexp)
            residual = hold['truth'][0] - mean
            white = np.linalg.solve(factor, residual[..., None])[..., 0]
            ll.ravel()[ids] = -.5*np.sum(white**2,axis=1)-np.log(np.diagonal(factor,axis1=1,axis2=2)).sum(axis=1)
            if start % 2048 == 0:
                print('CLOSURE new cells', start+len(batch), '/', len(missing), flush=True)
        assert np.isfinite(ll).all()
        return ll
    ll = evaluate(q, kappa)
    prior = infer._helper(infer.PRIOR_PATH)
    mass, tv = infer.convergence(ll,q,kappa)
    density = mass/(prior.weights(q)[:,None]*prior.weights(kappa)[None,:])
    # A real convergence failure is reported, not bypassed or silently refined.
    np.savez_compressed(ROOT / (output_tag+'.npz'), q=q,kappa=kappa,loglike=ll,mass=mass,density=density)
    true_u, true_v = hold['design_unit'][0]
    true_q, true_k = np.tan(np.pi*true_u/2)**3, .3+.3*true_v
    report = dict(fit_identity=identity, source_pins=pins, true_physical=[true_q,true_k],
        shape=list(ll.shape), coarse_fine_TV=float(tv), convergence_pass=bool(tv<.02),
        kind='One-point parameter recovery using heldout38 and full experimental covariance; no experimental replica draw. Includes measured target/train and target/target MC covariance.',
        caveat='One point is not repeated-sample coverage calibration or a posterior-local simulation test.')
    save(ROOT / ('CLOSURE_181x121.json' if refine else 'CLOSURE.json'), report)
    assert api.model_identity('all') == identity
    print('CLOSURE_COMPLETE', {k:report[k] for k in ('true_physical','shape','coarse_fine_TV','convergence_pass')}, flush=True)


def render_closure():
    import matplotlib.pyplot as plt
    import hybrid_plotting as hp
    import remaining_inference as infer
    from scipy.interpolate import RegularGridInterpolator
    suffix = '_181x121' if (ROOT/'CLOSURE_181x121.json').exists() else ''
    z, report = arrays(ROOT / ('closure_grid'+suffix+'.npz')), read(ROOT / ('CLOSURE'+suffix+'.json'))
    if not report['convergence_pass']:
        raise ValueError('Closure quadrature has not passed the stated convergence gate')
    prior = infer._helper(infer.PRIOR_PATH)
    q, k, mass, density = (z[key] for key in ('q','kappa','mass','density'))
    true_q, true_k = report['true_physical']
    ordered = np.argsort(density.ravel())[::-1]
    threshold = [density.ravel()[ordered][np.searchsorted(np.cumsum(mass.ravel()[ordered]),p)] for p in (.68,.95)]
    true_density = float(RegularGridInterpolator((q,k),density)([[true_q,true_k]])[0])
    report['physical_pi_T_Lres_16_50_84'] = prior.quantiles(mass,q,0)
    report['kappa_16_50_84'] = prior.quantiles(mass,k,1)
    report['truth_inside_68pct_joint_HPD'] = bool(true_density >= threshold[0])
    report['truth_inside_95pct_joint_HPD'] = bool(true_density >= threshold[1])
    with hp.paper_style():
        fig, axes = plt.subplots(1,2,figsize=(7.15,3.6),layout='constrained')
        for ax, x, marginal, truth, xlabel in (
            (axes[0],q,mass.sum(axis=1)/prior.weights(q),true_q,hp.LRES_LABEL),
            (axes[1],k,mass.sum(axis=0)/prior.weights(k),true_k,hp.KAPPA_LABEL)):
            ax.plot(x,marginal,color=hp.DEFAULT_COLOR,lw=2,label='Recovered posterior')
            ax.fill_between(x,0,marginal,color=hp.DEFAULT_COLOR,alpha=.15)
            ax.axvline(truth,color=hp.comparison_style('vermillion')['color'],ls='--',lw=1.5,label='Generating parameter')
            ax.set(xlim=(x[0],x[-1]),ylim=(0,None),xlabel=xlabel,ylabel='Posterior density')
            hp.frame_axes(ax,minor=False)
        axes[1].legend(loc='upper left',frameon=False,fontsize=8)
        hp.save_figure(fig, ROOT / 'qa_parameter_recovery')
        plt.close(fig)
    save(ROOT / 'CLOSURE_REVIEW.json',report)
    print('CLOSURE_RENDERED',{k:report[k] for k in ('true_physical','physical_pi_T_Lres_16_50_84','kappa_16_50_84','truth_inside_95pct_joint_HPD')},flush=True)
